## 1. 页面结构

#### 时间戳编辑操作

1. **手动标记**: 播放中按空格键标记当前时间
2. **自动对齐**: 调用后端 Silero VAD 自动检测
3. **拖动调整**: 拖动歌词时间轴调整
4. **预览效果**: 播放时高亮当前歌词

## 4. API 请求封装

### 4.1 Axios 实例配置

```javascript
// src/api/request.js
import axios from 'axios'

const api = axios.create({
  baseURL: '/api/v1',
  timeout: 30000,
  headers: {
    'Content-Type': 'application/json'
  }
})

// 请求拦截器
api.interceptors.request.use(config => {
  const token = localStorage.getItem('token')
  if (token) {
    config.headers.Authorization = `Bearer ${token}`
  }
  return config
})

// 响应拦截器
api.interceptors.response.use(
  response => response.data,
  error => {
    // 统一错误处理
    const message = error.response?.data?.message || '请求失败'
    return Promise.reject(new Error(message))
  }
)

export default api
```

### 4.2 API 模块封装

```javascript
// src/api/songs.js
import api from './request'

export const songsApi = {
  // 获取歌曲列表
  getList(params) {
    return api.get('/songs', { params })
  },
  
  // 上传歌曲
  upload(formData, onProgress) {
    return api.post('/songs', formData, {
      headers: { 'Content-Type': 'multipart/form-data' },
      onUploadProgress: onProgress
    })
  },
  
  // 删除歌曲
  delete(id) {
    return api.delete(`/songs/${id}`)
  },
  
  // 触发人声分离
  separate(id) {
    return api.post(`/songs/${id}/separate`)
  }
}
```

### 4.3 WebSocket 连接

```javascript
// src/api/websocket.js
import { io } from 'socket.io-client'

class WebSocketService {
  constructor() {
    this.socket = null
  }
  
  connect(taskId) {
    this.socket = io(`/ws/tasks/${taskId}`, {
      transports: ['websocket']
    })
    
    this.socket.on('progress', (data) => {
      // 更新进度
      this.onProgress?.(data)
    })
    
    this.socket.on('completed', (data) => {
      // 任务完成
      this.onCompleted?.(data)
    })
    
    this.socket.on('error', (error) => {
      // 错误处理
      this.onError?.(error)
    })
  }
  
  disconnect() {
    this.socket?.disconnect()
  }
}

export default new WebSocketService()
```

        const { data, total } = await songsApi.getList(this.pagination)
        this.list = data
        this.pagination.total = total
      } finally {
        this.loading = false
      }
    },
    
    async upload(formData) {
      const song = await songsApi.upload(formData)
      this.list.unshift(song)
      return song
    }
  }
})
